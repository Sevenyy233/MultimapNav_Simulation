cmake_minimum_required(VERSION 3.16)
project(clearing_robot_plugins)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(pluginlib REQUIRED)
find_package(multimap_nav_core REQUIRED)
find_package(multimap_nav_msgs REQUIRED)
find_package(multimap_nav_utils REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(tf2_eigen REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters kdtree registration)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(octomap REQUIRED)
find_package(ompl REQUIRED)
find_package(octomap_msgs REQUIRED)
find_package(grid_map_core REQUIRED)
find_package(grid_map_ros REQUIRED)
find_package(grid_map_msgs REQUIRED)
set(dependencies std_srvs rclcpp rclcpp_lifecycle pluginlib multimap_nav_core multimap_nav_msgs multimap_nav_utils tf2_ros tf2_geometry_msgs tf2_eigen geometry_msgs sensor_msgs nav_msgs std_msgs visualization_msgs pcl_conversions livox_ros_driver2 octomap_msgs grid_map_core grid_map_ros grid_map_msgs)
add_library(clearing_navigation SHARED src/model.cpp src/body_path.cpp src/map_geometry.cpp src/scene.cpp src/paper_global_planner.cpp src/hybrid_local_planner.cpp src/mppi_trajectory_planner.cpp src/localization.cpp src/recovery.cpp src/costmap2d_adapter.cpp src/gridmap_adapter.cpp)
target_include_directories(clearing_navigation PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${PCL_INCLUDE_DIRS})
target_include_directories(clearing_navigation PUBLIC ${OCTOMAP_INCLUDE_DIRS} ${OMPL_INCLUDE_DIRS})
target_link_libraries(clearing_navigation ${PCL_LIBRARIES} ${OCTOMAP_LIBRARIES} ${OMPL_LIBRARIES} yaml-cpp Eigen3::Eigen)
ament_target_dependencies(clearing_navigation ${dependencies})
add_executable(clearing_perception src/perception.cpp)
target_link_libraries(clearing_perception clearing_navigation)
ament_target_dependencies(clearing_perception ${dependencies})
add_executable(clearing_velocity_governor src/velocity_governor.cpp)
target_link_libraries(clearing_velocity_governor clearing_navigation)
ament_target_dependencies(clearing_velocity_governor ${dependencies})
pluginlib_export_plugin_description_file(multimap_nav_core plugins.xml)
ament_python_install_package(${PROJECT_NAME})
install(TARGETS clearing_navigation EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib)
install(TARGETS clearing_perception clearing_velocity_governor DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config launch behavior_trees rviz maps docs DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/prepare_assets.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(clearing_geometry_test test/test_geometry.cpp)
  target_compile_definitions(clearing_geometry_test PRIVATE CLEARING_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(clearing_geometry_test clearing_navigation)
  ament_target_dependencies(clearing_geometry_test ${dependencies})
endif()
ament_export_include_directories(include)
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(${dependencies} Eigen3 ompl)
ament_package()
