cmake_minimum_required(VERSION 3.8)
project(clearing_robot_simulation)
find_package(ament_cmake REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)

add_library(clearing_robot_contacts SHARED src/collision_monitor.cpp)
target_compile_features(clearing_robot_contacts PRIVATE cxx_std_17)
target_include_directories(clearing_robot_contacts SYSTEM PRIVATE ${GAZEBO_INCLUDE_DIRS})
ament_target_dependencies(clearing_robot_contacts gazebo_ros rclcpp std_msgs)
target_link_libraries(clearing_robot_contacts ${GAZEBO_LIBRARIES})
install(TARGETS clearing_robot_contacts LIBRARY DESTINATION lib)
install(DIRECTORY robot config launch worlds archive meshes 3DModel DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/initialize_clearing_arm.py scripts/clearing_tracks_drive.py scripts/wait_for_gazebo_clock.py DESTINATION lib/${PROJECT_NAME})
ament_package()
